394: /// A `DELETE/INSERT … WHERE` operation. Templates are ground or variable; the WHERE

477:         7 => format!(

647: /// (`op ; op`) to exercise within-one-request sequencing.

649: /// A `LOAD` is never compounded: its reference-engine substitute is a whole separate
650: /// request, and splicing two substitutes into one `;` chain would silently change the
651: /// sequencing the compound is there to test.
652: fn gen_op(rng: &mut Rng, st: &mut GenState) -> Op {
653:     fn single(ns: &str, rng: &mut Rng, st: &mut GenState) -> String {

765: /// The comparable form of one snapshot.
766: ///
767: /// With `relabel` false (neither side mentions a blank node) the sorted lines ARE the
768: /// canonical form and are returned untouched — the strongest, byte-level compare.
769: /// With it true, both sides are relabelled to their RDFC-1.0 canonical form so the
770: /// comparison decides RDF ISOMORPHISM. The constrained `*_ground_terms` profile is
771: /// used deliberately: it is exactly RDFC-1.0 with triple terms as opaque constants and
772: /// fails closed on a blank node nested inside a triple term, so the non-standard
773: /// nested-bnode descent is never reachable from this harness.
774: fn comparable(lines: &[String], relabel: bool, what: &str) -> Result<Vec<String>, String> {
775:     if !relabel {
776:         return Ok(lines.to_vec());
777:     }
778:     let quads = parse_lines(lines, what)?;
779:     let canon = sparq_canon::canonicalize_rdf12_ground_terms(&quads)
780:         .map_err(|e| format!("{}: RDFC-1.0 canonicalization failed ({})", what, e))?;
781:     let mut out: Vec<String> = canon.lines().map(str::to_string).collect();
782:     out.sort();
783:     Ok(out)
784: }

786: /// Rewrites `t` the way Oxigraph's storage layer does: an `xsd:integer` literal whose
787: /// value fits the native integer it decodes to is re-rendered in canonical lexical
788: /// form. Structural, never string surgery — and it descends into triple terms, which
789: /// Oxigraph's recursive term encoder also normalizes.
790: fn oxigraph_normalized_term(t: &Term) -> Term {
791:     match t {
792:         Term::Literal(l) if l.datatype() == xsd::INTEGER => match l.value().parse::<i64>() {
793:             Ok(v) if v.to_string() != l.value() => {
794:                 Term::Literal(Literal::new_typed_literal(v.to_string(), xsd::INTEGER))
795:             }
796:             _ => t.clone(),

851:     Same,
852:     /// Absorbed by the adjudicated `update-oxigraph-integer-lexical-canonicalization`
853:     /// class: the two datasets agree exactly once Oxigraph's numeric normalization is
854:     /// re-derived on both sides.
855:     AdjudicatedIntegerLexical,
856:     /// A real divergence (the string is the report body).
857:     Differs(String),
858: }

860: /// Compares two snapshots under the canonical form the module docs describe.
861: ///
862: /// `allow_integer_lexical` is set ONLY for a sparq-vs-Oxigraph compare, and only when
863: /// the allowlist enables the class. The sparq-vs-sparq compare passes false: both
864: /// sides are sparq, so any lexical disagreement between them is a real bug.
865: fn compare(
866:     label_a: &str,
867:     a: &[String],
868:     label_b: &str,
869:     b: &[String],
870:     allow_integer_lexical: bool,
871: ) -> Verdict {
872:     let relabel = mentions_blank_node(a) || mentions_blank_node(b);
873:     let (ca, cb) = match (
874:         comparable(a, relabel, label_a),
875:         comparable(b, relabel, label_b),
876:     ) {
877:         (Ok(ca), Ok(cb)) => (ca, cb),
878:         (Err(e), _) | (_, Err(e)) => return Verdict::Differs(e),
879:     };
880:     if ca == cb {
881:         // Canonicalization deduplicates, so a duplicate quad on one side alone would
882:         // survive the compare — check the raw counts to keep that failure visible.
883:         if a.len() != b.len() {
884:             return Verdict::Differs(format!(
885:                 "datasets are isomorphic but the raw quad COUNTS differ \
886:                  ({} {} vs {} {}) — one side is yielding a duplicate quad",
887:                 label_a,
888:                 a.len(),
889:                 label_b,
890:                 b.len()
891:             ));
892:         }
893:         return Verdict::Same;
894:     }
895:     if allow_integer_lexical {
896:         let normalized = (
897:             oxigraph_normalized(a, label_a).and_then(|n| comparable(&n, relabel, label_a)),
898:             oxigraph_normalized(b, label_b).and_then(|n| comparable(&n, relabel, label_b)),
899:         );
900:         if let (Ok(na), Ok(nb)) = normalized {
901:             if na == nb {
902:                 return Verdict::AdjudicatedIntegerLexical;
903:             }
904:         }
905:     }
906:     Verdict::Differs(one_sided(label_a, &ca, label_b, &cb))
907: }
