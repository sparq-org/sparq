255:     /// How many overlay triples fall in the `[lo, hi]` range of `perm` — the exact
256:     /// correction to a base range count. The `added` side rides the cached perm-sorted
257:     /// projection (O(log k), two binary searches); the `deleted` side is an unordered

259:     fn count_correction(&self, perm: Perm, lo: [Id; 3], hi: [Id; 3]) -> (usize, usize) {
260:         let order = perm.order();

262:         let del = self
263:             .deleted
264:             .iter()
265:             .filter(|t| {
266:                 let r = [t[order[0]], t[order[1]], t[order[2]]];
267:                 r >= lo && r <= hi
268:             })
269:             .count();
270:         (add, del)
271:     }
