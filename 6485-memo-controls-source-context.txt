267:     /// new terms recorded in a per-fork side map. Forking a fork shares the same
268:     /// base Arc and copies the (small) side map; the FIRST fork of a flat cache

809:     let s = s.trim();

811:     let (int, frac) = s.split_once('.').unwrap_or((s, ""));

814:     }

816:     let frac = frac.trim_end_matches('0');
817:     if int.is_empty() {
818:         frac.trim_start_matches('0').len()
819:     } else {
820:         int.len() + frac.len()
821:     }
822: }
